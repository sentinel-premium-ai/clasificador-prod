to_char(ic.created_at,'dd/mm/yyyy') fecha,
count(ic.file_name) fotografías
from neuralsmf."image_classification" ic
where to_char(ic.created_at,'yyyymm') = '202502'
group by to_char(ic.created_at,'dd/mm/yyyy')
order by to_char(ic.created_at,'dd/mm/yyyy') desc;
